// JavaScript / Fetch // Méthode GET — retourne l'image brute
const imageUrl = "https://ton-image.jpg";
const prompt = "change le fond en plage";

const url = `/api/img?url=${encodeURIComponent(imageUrl)}&prompt=${encodeURIComponent(prompt)}`;

const res = await fetch(url);
const blob = await res.blob();
const src = URL.createObjectURL(blob);
img.src = src;
// Python / requests # Méthode POST — retourne JSON + base64
import requests

res = requests.post("/api/img", json={
  "imageUrl": "https://ton-image.jpg",
  "prompt": "change le fond en plage"
})

data = res.json()
# data["dataUrl"] → data:image/jpeg;base64,...
